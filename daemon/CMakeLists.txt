# hubusd: MQTT master daemon (docs/DAEMON_REQUIREMENTS.md).
# Needs libyaml for the configuration; the MQTT client (libmosquitto) and the
# libgpiod HAL are only needed for the executable itself.
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
    pkg_check_modules(YAML QUIET yaml-0.1)
    pkg_check_modules(MOSQUITTO QUIET libmosquitto)
endif()

if(YAML_FOUND)
    add_library(hubusd_lib STATIC
        hubusd_yaml.c hubusd_config.c hubusd_profile.c hubusd_json.c hubusd_core.c)
    target_include_directories(hubusd_lib PUBLIC . ${YAML_INCLUDE_DIRS})
    target_link_libraries(hubusd_lib PUBLIC hubus_master ${YAML_LIBRARIES} pthread)
    target_compile_options(hubusd_lib PRIVATE ${HUBUS_WARNINGS})
    set(HUBUSD_LIB_AVAILABLE TRUE PARENT_SCOPE)
    message(STATUS "libyaml ${YAML_VERSION} found: building the daemon library")
else()
    set(HUBUSD_LIB_AVAILABLE FALSE PARENT_SCOPE)
    message(STATUS "libyaml not found: skipping hubusd")
endif()

if(YAML_FOUND AND MOSQUITTO_FOUND)
    add_library(hubusd_run STATIC hubusd_run.c)
    target_include_directories(hubusd_run PUBLIC . ${MOSQUITTO_INCLUDE_DIRS})
    target_link_libraries(hubusd_run PUBLIC hubusd_lib ${MOSQUITTO_LIBRARIES})
    target_compile_options(hubusd_run PRIVATE ${HUBUS_WARNINGS})
    if(TARGET hubus_master_gpiod)
        add_executable(hubusd main.c)
        target_link_libraries(hubusd PRIVATE hubusd_run hubus_master_gpiod)
        target_compile_options(hubusd PRIVATE ${HUBUS_WARNINGS})
        install(TARGETS hubusd RUNTIME DESTINATION bin)
    endif()
    install(FILES hubusd.example.yaml DESTINATION share/hubus)
    install(FILES hubusd.service DESTINATION lib/systemd/system)
    message(STATUS "libmosquitto ${MOSQUITTO_VERSION} found: building the hubusd executable")
else()
    message(STATUS "libmosquitto not found: hubusd executable not built")
endif()
